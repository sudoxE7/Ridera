const { onValueCreated, onValueWritten } = require("firebase-functions/v2/database");
const admin = require("firebase-admin");
const axios = require("axios");

admin.initializeApp();

// ======================================
// PH TIME
// ======================================

function getCurrentTimePH() {

  return new Date()
    .toLocaleTimeString(
      "en-PH",
      {
        hour12: false,
        timeZone: "Asia/Manila"
      }
    );
}

function getCurrentDatePH() {

  return new Date()
    .toLocaleDateString(
      "en-PH",
      {
        timeZone: "Asia/Manila"
      }
    );
}

exports.sendCrashNotification = onValueCreated(
"/Ridera/devices/{deviceId}/telematics/crash",

async (event) => {

  console.log("FUNCTION TRIGGERED");

  const crash = event.data.val();

  console.log("CRASH DATA:", crash);

  if (!crash) {

    console.log("NO CRASH DATA");

    return;
  }

  const deviceId =
    event.params.deviceId;

  console.log(
    "DEVICE ID:",
    deviceId
  );

  // ======================================
  // FALLBACK LIVE LOCATION
  // ======================================

  let fallbackLocation = null;

  try {

    const locationSnap =
      await admin.database()
        .ref(
          `/Ridera/devices/${deviceId}/telematics/location`
        )
        .once("value");

    fallbackLocation =
      locationSnap.val();

    console.log(
      "FALLBACK LOCATION:",
      fallbackLocation
    );

  } catch (err) {

    console.log(
      "FALLBACK LOCATION ERROR:",
      err
    );
  }

  // ======================================
  // GET BINDING UID
  // ======================================

  const deviceSnap =
    await admin.database()
      .ref(
        `/Ridera/devices/${deviceId}/binding/uid`
      )
      .once("value");

  const authUid =
    deviceSnap.val();

  console.log(
    "AUTH UID:",
    authUid
  );

  if (!authUid) {

    console.log(
      "NO BINDING UID FOUND"
    );

    return;
  }

  // ======================================
  // FIND USER
  // ======================================

  const usersSnap =
    await admin.database()
      .ref(`/Ridera/users`)
      .orderByChild("uid")
      .equalTo(authUid)
      .once("value");

  if (!usersSnap.exists()) {

    console.log(
      "USER NOT FOUND"
    );

    return;
  }

  let userKey = null;
  let userData = null;

  usersSnap.forEach(child => {

    userKey =
      child.key;

    userData =
      child.val();
  });

  console.log(
    "USER KEY:",
    userKey
  );

  // ======================================
  // FALLBACK USER DATA
  // ======================================

  const finalPhoto =
    crash.photo ||
    userData.photo ||
    "";

  const finalName =
    crash.name ||
    userData.name ||
    "";

  const finalSex =
    crash.sex ||
    userData.sex ||
    "";

  const finalPhone =
    crash.phone ||
    userData.phone ||
    "";

  const finalVehicleType =
    crash.vehicle_type ||
    userData.vehicle_type ||
    "";

  const finalVehicleModel =
    crash.vehicle_model ||
    userData.vehicle_model ||
    "";

  const finalVehicleColor =
    crash.vehicle_color ||
    userData.vehicle_color ||
    "";

  const finalVehiclePlate =
    crash.vehicle_plate ||
    userData.vehicle_plate ||
    "";

  // ======================================
  // GPS FALLBACK
  // ======================================

  const invalidGps = (

    !crash.latitude ||
    !crash.longitude ||

    crash.latitude == 0 ||
    crash.longitude == 0
  );

  const finalLatitude =
    invalidGps

    ? (
        fallbackLocation?.latitude > 0
        ? fallbackLocation.latitude
        : 0
      )

    : crash.latitude;

  const finalLongitude =
    invalidGps

    ? (
        fallbackLocation?.longitude > 0
        ? fallbackLocation.longitude
        : 0
      )

    : crash.longitude;

  // ======================================
  // SPEED FALLBACK
  // ======================================

  const invalidSpeed = (

    crash.speed == null ||

    isNaN(crash.speed) ||

    crash.speed <= 0
  );

  const finalSpeed =
    invalidSpeed

  ? (
      fallbackLocation?.speed_kmph > 0
      ? fallbackLocation.speed_kmph
      : 0
    )

  : crash.speed;

  // ======================================
  // TIME FALLBACK
  // ======================================

  const invalidTime = (
    !crash.time ||
    !crash.date ||
    
    crash.time === "08:00:00" ||
    crash.date === "00:00:2000" ||
    crash.date === "2000-00-00"
  );

  const finalDate =
    invalidTime
    ? getCurrentDatePH()
    : crash.date;

  const finalTime =
    invalidTime
    ? getCurrentTimePH()
    : crash.time;

  // ======================================
  // GET FCM TOKEN
  // ======================================

  const tokenSnap =
    await admin.database()
      .ref(
        `/Ridera/users/${userKey}/fcmToken`
      )
      .once("value");

  const token =
    tokenSnap.val();

  console.log(
    "FCM TOKEN:",
    token
  );

  if (!token) {

    console.log(
      "NO TOKEN FOUND"
    );

    return;
  }

  // ======================================
  // CRASH HISTORY
  // ======================================

  const historyRoot =
    admin.database()
      .ref(
        `/Ridera/users/${userKey}/crash_alerts`
      );

  const snap =
    await historyRoot.once("value");

  const count =
    snap.numChildren() + 1;

  const crashId =
    "crash_" +
    String(count).padStart(3, "0");

  console.log(
    "NEW CRASH ID:",
    crashId
  );

  const crashRef =
    historyRoot.child(crashId);

  // ======================================
  // SAVE HISTORY
  // ======================================

  let priority = "LOW";

  if (finalSpeed >= 20) {
    priority = "HIGH";
  }
  else if (finalSpeed >= 5) {
    priority = "MEDIUM";
  }

  if (crash.severity === "medium") {
    priority = "MEDIUM";
  }

  if (crash.severity === "high") {
    priority = "HIGH";
  }

  await crashRef.set({

    ...crash,

    speed:
      finalSpeed,

    latitude:
      finalLatitude,

    longitude:
      finalLongitude,

    date:
      finalDate,

    time:
      finalTime,

    photo:
      finalPhoto,

    name:
      finalName,

    sex:
      finalSex,

    phone:
      finalPhone,

    vehicle_type:
      finalVehicleType,

    vehicle_model:
      finalVehicleModel,

    vehicle_color:
      finalVehicleColor,

    vehicle_plate:
      finalVehiclePlate,

    crashId:
      crashId,

    userId: 
      userKey,

    createdAt: 
      Date.now(),

    incident_status:
      "active",

    priority:
      priority,

    responder_agency: "CDRRMO Dasmarinas",

    rider_status: "NO_RESPONSE",

    rider_note: "",

    updatedAt: Date.now(),

    dispatch_recommended:
      priority === "HIGH",

    alarm_type:
      priority === "HIGH"
        ? "continuous"
        : priority === "MEDIUM"
        ? "repeating"
        : "single",
    });
    console.log("CRASH SAVED");

  // ======================================
  // SEND NOTIFICATION
  // ======================================

  let notificationTitle = "Emergency Alert";

  switch (crash.type) {
    case "motorcycle_crash":
      notificationTitle =
        "Motorcycle Crash Detected";
      break;

    case "motorcycle_fall":
      notificationTitle =
        "Motorcycle Fall Detected";
      break;

    case "stationary_impact":
      notificationTitle =
        "Stationary Impact Detected";
      break;

    default:
      notificationTitle =
        "Emergency Alert";
  }

  const payload = {

    token: token,

    data: {

      title: notificationTitle,

      body: `${priority} Priority Incident Detected`,

      crashId:
        crashId,

      userId:
        userKey,

      type:
        crash.type || "",

      severity:
        crash.severity || "",

      speed:
        String(
          finalSpeed
        ),

      lat:
        String(finalLatitude),

      lng:
        String(finalLongitude),

      time:
        finalTime,

      date:
        finalDate,

      photo:
        finalPhoto,

      name:
        finalName,

      sex:
        finalSex,

      phone:
        finalPhone,

      vehicle_type:
        finalVehicleType,

      vehicle_model:
        finalVehicleModel,

      vehicle_color:
        finalVehicleColor,

      vehicle_plate:
        finalVehiclePlate,

      seen:
        "false",

        priority:
  priority,

alarm_type:
  priority === "HIGH"
    ? "continuous"
    : priority === "MEDIUM"
    ? "repeating"
    : "single",
    }
  };

  try {

    const response =
      await admin.messaging()
        .send(payload);

    console.log(
      "NOTIFICATION SENT:",
      response
    );

  } catch (err) {

    console.log(
      "FCM ERROR:",
      err
    );
  }

  // ======================================
  // SEND SMS TO CONTACTS
  // ======================================
    try {

      const latestSnap =
        await crashRef.once("value");

      const latestData =
        latestSnap.val();

        // ======================================
        // GET CONTACTS
        // ======================================

        const contactsSnap =
          await admin.database()
            .ref(
              `/Ridera/users/${userKey}/alert_contacts`
            )
            .once("value");

        let numbers = [];

        // ======================================
        // VEHICLE INCIDENT CHECK
        // ======================================

        const isVehicleOnly = false;

        if (isVehicleOnly) {
          // owner only
          if (finalPhone) {
            numbers.push(String(finalPhone).trim());
          }
        } else {
          // owner + alert contacts
          if (finalPhone) {
            numbers.push(String(finalPhone).trim());
          }

          if (contactsSnap.exists()) {
            contactsSnap.forEach(child => {
              const number = child.child("contact_phone").val();

              if (number) {
                numbers.push(String(number).trim());
              }
            });
          }
        }

        console.log(
          "CONTACTS:",
          numbers
        );

        numbers = [...new Set(numbers)];

        // ======================================
        // FORMAT TYPE
        // ======================================

        let formattedType = "";

        switch (latestData.type) {

          case "motorcycle_crash":
            formattedType = "Motorcycle Crash";
            break;

          case "motorcycle_fall":
            formattedType = "Motorcycle Fall";
            break;

          case "stationary_impact":
            formattedType = "Stationary Impact";
            break;

          default:
            formattedType = "Emergency Alert";
        } 

        // ======================================
        // BUILD SMS
        // ======================================

        let sms = "";

        sms +=
          "EMERGENCY ALERT!\n\n";

        sms +=
          "Type: " +
          (formattedType) +
          "\n";

        sms +=
          "Severity: " +
          (latestData.severity || "") +
          "\n";

          sms +=
  "Priority: " +
  priority +
  "\n";

        sms +=
          "Speed: " +
          String(
            finalSpeed || 0
          ) +
          " km/h\n";

        sms +=
          "Location:\nhttps://maps.google.com/?q=" +
          String(
            latestData.latitude || 0
          ) +
          "," +
          String(
            latestData.longitude || 0
          ) +
          "\n";

        sms +=
          "Time: " +
          (latestData.time || "") +
          " | " +
          (latestData.date || "") +
          "\n\n";

        // ======================================
        // INCIDENT NOTE
        // ======================================
        if (isVehicleOnly) {
          sms +=
          "NOTE:\n" +
          "Please check your vehicle immediately for any visible damages or safety issues.\n\n";
        } else {
          sms +=
          "NOTE:\n" +
          "Possible rider injury detected. Emergency contacts have been notified.\n\n";
        }

        // ======================================
        // RIDER INFO
        // ======================================

        if (
          latestData.type === "motorcycle_crash" ||
          latestData.type === "stationary_impact" ||
          latestData.type === "motorcycle_fall"
        ) {

          sms +=
            "RIDER INFORMATION\n";

          sms +=
            "Name: " +
            finalName +
            "\n";

          sms +=
            "Phone: " +
            finalPhone +
            "\n";

          sms +=
            "Vehicle Type: " +
            finalVehicleType +
            "\n";

          sms +=
            "Vehicle Model: " +
            finalVehicleModel +
            "\n";

          sms +=
            "Vehicle Color: " +
            finalVehicleColor +
            "\n";

          sms +=
            "Plate No: " +
            finalVehiclePlate +
            "\n";
        }

        console.log("SMS:");
        console.log(sms);
        /*
        // ======================================
        // SEND SMS
        // ======================================

          try {

            const response =
              await axios.post(

                "https://semaphore.co/api/v4/priority",

                new URLSearchParams({

                  apikey:
                    "962ba2681080eba8cfb9073f4b4f7eb5",

                  number:
                    numbers.join(","),

                  message:
                    sms
                }),

                {
                  headers: {
                    "Content-Type":
                      "application/x-www-form-urlencoded"
                  }
                }
              );

            console.log(
              "SMS SUCCESS:",
              response.data
            );

          } catch (smsErr) {

            console.log(
              "SMS ERROR:",
              smsErr.response?.data ||
              smsErr.message
            );
          }
        }
          */

    } catch (err) {

      console.log(
        "AUTO SEND ERROR:",
        err
      );
    }

  // ======================================
  // DELETE LIVE CRASH
  // ======================================

  await admin.database()
    .ref(
      `/Ridera/devices/${deviceId}/telematics/crash`
    )
    .remove();

  console.log(
    "LIVE CRASH DELETED"
  );

  return null;
});

exports.sendIncidentStatusNotification =
onValueWritten(
"/Ridera/users/{userId}/crash_alerts/{crashId}/incident_status",

async (event) => {

  const before =
    event.data.before.val();

  const after =
    event.data.after.val();

  if (!after || before === after) {
    return null;
  }

  const userId =
    event.params.userId;

  const crashId =
    event.params.crashId;

  const tokenSnap =
    await admin.database()
      .ref(
        `/Ridera/users/${userId}/fcmToken`
      )
      .once("value");

  const token =
    tokenSnap.val();

  if (!token) {
    return null;
  }

  let title = "";
  let body = "";

  switch (after) {

  case "active":
    return null;

  case "acknowledged":
    title = "Incident Acknowledged";
    body =
      "CDRRMO has acknowledged your incident report.";
    break;

  case "dispatched":
    title = "Response Team Dispatched";
    body =
      "Emergency responders have been dispatched.";
    break;

  case "arrived":
    title = "Responders Arrived";
    body =
      "Emergency responders have arrived at your location.";
    break;

  case "resolved":
    title = "Incident Resolved";
    body =
      "This incident has been resolved.";
    break;

  default:
    return null;
}

  await admin.messaging().send({

    token,

    data: {

      title,
      body,

      type:
        "responder_status",

      status:
        after,

      crashId:
        crashId
    }
  });

  console.log(
    "STATUS NOTIFICATION SENT:",
    after
  );

  return null;
});
